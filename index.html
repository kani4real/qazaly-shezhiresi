"""

from pathlib import Path

import pandas as pd
import matplotlib.pyplot as plt

PAPKA = Path(__file__).resolve().parent.parent

NATIZHE.mkdir(exist_ok=True)

zhiyntyq = df.pivot_table(index="Мәселе", columns="Дереккөз",
                          values="Көрсетілген жыл", aggfunc="first")

# 3. Әр мәселе бойынша қысқаша статистика
stat = df.groupby("Мәселе")["Көрсетілген жыл"].agg(
    sany="count",
    erte="min",
    kesh="max",
    zhii=lambda s: s.mode().iloc[0],
)
stat.columns = ["Дереккөз саны", "Ең ерте жыл", "Ең кеш жыл", "Жиі кездесетін жыл"]
stat["Айырма (жыл)"] = stat["Ең кеш жыл"] - stat["Ең ерте жыл"]


# 4. «Дата дауы» блогы: әр жылдың қасына оны көрсеткен дереккөздерді жазу
def zhyldar_men_derekkozder(top):
    zholdar = []
    for zhyl, bolik in top.groupby("Көрсетілген жыл"):
        zholdar.append(f"{zhyl} – {', '.join(bolik['Дереккөз'])}")
    return " | ".join(zholdar)


blok = {masele: zhyldar_men_derekkozder(top) for masele, top in df.groupby("Мәселе")}
stat["Дата дауы"] = pd.Series(blok)

# Мәселелерді кестедегі ретімен орналастыру
stat = stat.reindex(list(dict.fromkeys(df["Мәселе"])))

print("\n«ДАТА ДАУЫ» – ҚЫСҚАША СТАТИСТИКА")
print(stat[["Дереккөз саны", "Ең ерте жыл", "Ең кеш жыл",
            "Жиі кездесетін жыл", "Айырма (жыл)"]].to_string())

# 5. Нәтижені Excel-ге сақтау
with pd.ExcelWriter(NATIZHE / "data_dauy_natizhe.xlsx", engine="openpyxl") as w:
    stat.to_excel(w, sheet_name="Статистика")
    zhiyntyq.to_excel(w, sheet_name="Жиынтық_кесте")
    df.to_excel(w, sheet_name="Бастапқы_деректер", index=False)

# 6. Диаграмма: әр мәселе – жеке сызба, нүктелер – дереккөздер берген жылдар
maseleler = list(stat.index)
tus = ["#1f4e79", "#c0504d", "#4f9a3a", "#8064a2", "#f79646", "#4bacc6"]
fig, axes = plt.subplots(len(maseleler), 1, figsize=(11, 2.4 * len(maseleler)))

for ax, masele in zip(axes, maseleler):
    top = df[df["Мәселе"] == masele]
    erte, kesh = top["Көрсетілген жыл"].min(), top["Көрсетілген жыл"].max()
    ax.hlines(0, erte, kesh, color="#cccccc", linewidth=8, zorder=1)
    for j, (zhyl, bolik) in enumerate(top.groupby("Көрсетілген жыл")):

        ax.annotate("\n".join(bolik["Дереккөз"]), (zhyl, -0.45), ha="center",

    ax.set_title(f"{masele}  (айырма: {kesh - erte} жыл)", fontsize=11,

    ax.set_xlim(erte - 1.2, kesh + 1.2)
    ax.set_ylim(-1.3, 0.5)
    ax.set_yticks([])
    ax.xaxis.set_major_locator(plt.MultipleLocator(1))
    ax.tick_params(axis="x", labelsize=9)
    for jaq in ["top", "right", "left"]:
        ax.spines[jaq].set_visible(False)

fig.suptitle("«Дата дауы»: бір оқиғаға әр дереккөз берген даталар",

plt.tight_layout()
plt.savefig(NATIZHE / "data_dauy_diagramma.png", dpi=200)
print("\nСақталды:", NATIZHE / "data_dauy_natizhe.xlsx")

def bos_emes(x):
    """Ұяшық бос емес пе екенін тексереді."""
    return pd.notna(x) and str(x).strip() != ""


def kun(zhyl, ai=None, kun_=None):
    d = {"year": int(zhyl)}
    if bos_emes(ai):
        d["month"] = int(ai)
    if bos_emes(kun_):
        d["day"] = int(kun_)
    return d

def qashyqtyq_km(lat1, lon1, lat2, lon2):
    """Екі нүкте арасындағы қашықтық (гаверсинус формуласы), км."""
    R = 6371
    f1, f2 = math.radians(lat1), math.radians(lat2)
    df = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)
    a = math.sin(df / 2) ** 2 + math.cos(f1) * math.cos(f2) * math.sin(dl / 2) ** 2
    return 2 * R * math.asin(math.sqrt(a))

    # -*- coding: utf-8 -*-
"""
4-қадам. Стендке арналған QR-код.

1-нұсқа: QR-кодқа зерттеудің қысқаша нәтижесі жазылады (интернетсіз оқылады).
2-нұсқа: хронология интернетке жарияланған соң, оның сілтемесін SILTEME
айнымалысына жазып, скриптті қайта іске қосу жеткілікті.

Іске қосу:  python3 04_qr_kod.py
Қажетті кітапхана:  qrcode[pil]
"""
from pathlib import Path

import qrcode
from PIL import Image, ImageDraw, ImageFont

PAPKA = Path(__file__).resolve().parent.parent
NATIZHE = PAPKA / "5_Нәтижелер"

SILTEME = "https://kani4real.github.io/qazaly-shezhiresi/"  # хронологияның интернеттегі мекенжайы

MATIN = (
    "ҚАЗАЛЫ ШЕЖІРЕСІ (1847–1906)\n"
    "1847 – Райым бекінісі салынды\n"
    "1853 – №1 форт (Қазалы) салынды\n"
    "1854 – Сырдария шебінде ресми тіркелді\n"
    "1855 – Райым гарнизоны №1 фортқа көшті\n"
    "1867 – Казалинск қала мәртебесін алды, уезд орталығы болды\n"
    "1905 – Қазалы вокзалы салынды\n"
    "Қорытынды: Қазалының құрылған жылы – 1853."
)

mazmun = SILTEME if SILTEME else MATIN

qr = qrcode.QRCode(error_correction=qrcode.constants.ERROR_CORRECT_M, box_size=10, border=3)
qr.add_data(mazmun)
qr.make(fit=True)
kod = qr.make_image(fill_color="#1f4e79", back_color="white").convert("RGB")

# QR-кодтың астына жазу қосу
eni, biiktigi = kod.size
stend = Image.new("RGB", (eni, biiktigi + 70), "white")
stend.paste(kod, (0, 0))
draw = ImageDraw.Draw(stend)
try:
    font = ImageFont.truetype("Arial.ttf", 26)
except OSError:
    font = ImageFont.load_default()
zhazu = "Форт №1-ден Қазалыға дейін"
w = draw.textlength(zhazu, font=font)
draw.text(((eni - w) / 2, biiktigi + 15), zhazu, fill="#1f4e79", font=font)

NATIZHE.mkdir(exist_ok=True)
stend.save(NATIZHE / "qr_kod.png")
print("QR-код нұсқасы:", qr.version, "| таңба саны:", len(mazmun))
print("Сақталды:", NATIZHE / "qr_kod.png")
